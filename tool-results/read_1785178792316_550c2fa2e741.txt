   199→    // 4. LISTAR CITAS (Con filtros de fecha, estado y datos de pago)
   200→    public function leer($fecha_desde = null, $fecha_hasta = null, $estado = null, $id_medico = null, $id_paciente = null) {
   201→        $query = 'SELECT c.id_cita, c.fecha_cita, c.motivo, c.estado, c.id_medico, c.id_paciente, c.id_servicio,
   202→                         c.diagnostico, c.prescripcion, c.tratamiento,
   203→                         c.peso, c.talla, c.temperatura, c.presion_arterial,
   204→                         c.fc, c.fr, c.sat_o2, c.dias_reposo, c.fecha_fin_reposo,

   206→                         c.antecedentes, c.examen_fisico, c.examenes_auxiliares,
   207→                         c.fecha_control, c.id_interconsulta_especialidad,
   208→                         u.nombre as paciente, u.email as paciente_email, u.telefono as paciente_telefono,
   209→                         u.documento_identidad, u.fecha_nacimiento, u.sexo,
   210→                         u.persona_responsable, u.tel_responsable, u.nro_historia_clinica,
   211→                         m_u.nombre as medico, e.nombre as especialidad, m.colegiatura,
   212→                         s.nombre_servicio, s.precio,
   213→                         p.id_pago, p.metodo_pago, p.fecha_pago, p.monto as monto_pagado
   214→                  FROM ' . $this->table . ' c
   215→                  JOIN usuarios u ON c.id_paciente = u.id_usuario
   216→                  JOIN medicos m ON c.id_medico = m.id_medico
   217→                  JOIN usuarios m_u ON m.id_usuario = m_u.id_usuario
   218→                  JOIN especialidades e ON m.id_especialidad = e.id_especialidad
   219→                  LEFT JOIN servicios s ON c.id_servicio = s.id_servicio
   220→                  LEFT JOIN pagos p ON c.id_cita = p.id_cita'; 
   221→        
   222→        $condiciones = [];
   223→        
   224→        // Filtro por rango de fechas
   225→        if ($fecha_desde && $fecha_hasta) {
   226→            $condiciones[] = 'DATE(c.fecha_cita) BETWEEN :desde AND :hasta';
   227→        } elseif ($fecha_desde) {
   228→            $condiciones[] = 'DATE(c.fecha_cita) >= :desde';
   229→        }
   230→
   231→        if ($estado) $condiciones[] = 'c.estado = :estado';
   232→        if ($id_medico) $condiciones[] = 'c.id_medico = :medico_id';
   233→        if ($id_paciente) $condiciones[] = 'c.id_paciente = :paciente_id';
   234→
   235→        if (count($condiciones) > 0) $query .= ' WHERE ' . implode(' AND ', $condiciones);
   236→        $query .= ' ORDER BY c.fecha_cita ASC';
   237→        
   238→        $stmt = $this->conn->prepare($query);
   239→        
   240→        if ($fecha_desde && $fecha_hasta) {
   241→            $stmt->bindParam(':desde', $fecha_desde);
   242→            $stmt->bindParam(':hasta', $fecha_hasta);
   243→        } elseif ($fecha_desde) {
   244→            $stmt->bindParam(':desde', $fecha_desde);
   245→        }
   246→
   247→        if ($estado) $stmt->bindParam(':estado', $estado);
   248→        if ($id_medico) $stmt->bindParam(':medico_id', $id_medico);
   249→        if ($id_paciente) $stmt->bindParam(':paciente_id', $id_paciente);
   250→        
   251→        $stmt->execute();
   252→        return $stmt;
   253→    }

   655→    // 3. VERIFICAR HORARIO LABORAL
   656→    // Primero revisa si hay un horario específico para esa fecha exacta.
   657→    // Si no hay, cae al horario recurrente por día de la semana.
   658→    public function verificaHorarioLaboral($id_medico, $fecha_hora) {
   659→        $dias = [
   660→            'Monday' => 'Lunes', 'Tuesday' => 'Martes', 'Wednesday' => 'Miércoles',
   661→            'Thursday' => 'Jueves', 'Friday' => 'Viernes', 'Saturday' => 'Sábado', 'Sunday' => 'Domingo'
   662→        ];
   663→
   664→        $timestamp  = strtotime($fecha_hora);
   665→        $dia_ingles = date('l', $timestamp);
   666→        $dia_es     = $dias[$dia_ingles] ?? '';
   667→        $hora_cita  = date('H:i:s', $timestamp);
   668→        $fecha_cita = date('Y-m-d', $timestamp);
   669→
   670→        // 1. ¿Existe horario ESPECÍFICO para exactamente esa fecha?
   671→        $sqlEspecifico = "SELECT COUNT(*) FROM horarios_medicos 
   672→                          WHERE id_medico = :id 
   673→                          AND fecha_especifica = :fecha 
   674→                          AND :hora >= hora_inicio 
   675→                          AND :hora < hora_fin";
   676→        $stmtE = $this->conn->prepare($sqlEspecifico);
   677→        $stmtE->bindParam(':id',    $id_medico);
   678→        $stmtE->bindParam(':fecha', $fecha_cita);
   679→        $stmtE->bindParam(':hora',  $hora_cita);
   680→        $stmtE->execute();
   681→
   682→        if ($stmtE->fetchColumn() > 0) {
   683→            return true; // Hay turno específico para ese día → válido
   684→        }
   685→
   686→        // 2. ¿Existe algún horario ESPECÍFICO para ese mismo día (fecha)?
   687→        //    Si existe pero no coincide la hora, bloqueamos (no caemos al recurrente)
   688→        $sqlTieneEspecifico = "SELECT COUNT(*) FROM horarios_medicos 
   689→                               WHERE id_medico = :id 
   690→                               AND fecha_especifica = :fecha";
   691→        $stmtT = $this->conn->prepare($sqlTieneEspecifico);
   692→        $stmtT->bindParam(':id',    $id_medico);
   693→        $stmtT->bindParam(':fecha', $fecha_cita);
   694→        $stmtT->execute();
   695→
   696→        if ($stmtT->fetchColumn() > 0) {
   697→            return false; // Hay turnos específicos ese día pero ninguno cubre la hora
   698→        }
   699→
   700→        // 3. Sin horario específico: usar el horario RECURRENTE del día de la semana
   701→        $sqlRecurrente = "SELECT COUNT(*) FROM horarios_medicos 
   702→                          WHERE id_medico = :id 
   703→                          AND dia_semana = :dia 
   704→                          AND fecha_especifica IS NULL
   705→                          AND :hora >= hora_inicio 
   706→                          AND :hora < hora_fin";
   707→        $stmtR = $this->conn->prepare($sqlRecurrente);
   708→        $stmtR->bindParam(':id',  $id_medico);
   709→        $stmtR->bindParam(':dia', $dia_es);
   710→        $stmtR->bindParam(':hora', $hora_cita);

   714→    }
